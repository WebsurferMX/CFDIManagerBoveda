@code {
    private CfdiFiltroDto Filtro { get; set; } = new()
    {
        FechaInicio = new DateTime(2024, 1, 1),
        FechaFin = DateTime.Today,
        Pagina = 1,
        RegistrosPorPagina = 25
    };

    private ResultadoPaginadoDto<CfdiConsultaDto> Resultado { get; set; } = new();
    private CfdiResumenConsultaDto Resumen { get; set; } = new();
    private CfdiConsultaDto? Seleccionado { get; set; }
    private bool Cargando { get; set; } = true;
    private string? Error { get; set; }

    protected override void OnInitialized()
    {
        CargarDatos();
    }

    private void Consultar()
    {
        Filtro.Pagina = 1;
        Seleccionado = null;
        CargarDatos();
    }

    private void CargarDatos()
    {
        Cargando = true;
        Error = null;

        try
        {
            Resultado = Repo.BuscarCfdi(Filtro);
            Resumen = Repo.ObtenerResumenCfdi(Filtro);
        }
        catch (Exception ex)
        {
            Error = ex.Message;
        }
        finally
        {
            Cargando = false;
        }
    }

    private void LimpiarFiltros()
    {
        Filtro = new CfdiFiltroDto
        {
            FechaInicio = new DateTime(2024, 1, 1),
            FechaFin = DateTime.Today,
            Pagina = 1,
            RegistrosPorPagina = 25
        };
        Seleccionado = null;
        CargarDatos();
    }

    private void AplicarTipo(string tipo)
    {
        Filtro.TipoCfdi = tipo;
        Filtro.Pagina = 1;
        Seleccionado = null;
        CargarDatos();
    }

    private void Seleccionar(CfdiConsultaDto cfdi)
    {
        Seleccionado = cfdi;
    }

    private void PaginaAnterior()
    {
        if (!Resultado.TienePaginaAnterior) return;
        Filtro.Pagina--;
        Seleccionado = null;
        CargarDatos();
    }

    private void PaginaSiguiente()
    {
        if (!Resultado.TienePaginaSiguiente) return;
        Filtro.Pagina++;
        Seleccionado = null;
        CargarDatos();
    }

    private void CambiarTamanoPagina(ChangeEventArgs args)
    {
        if (int.TryParse(args.Value?.ToString(), out int cantidad))
        {
            Filtro.RegistrosPorPagina = cantidad;
            Filtro.Pagina = 1;
            CargarDatos();
        }
    }

    private void Ordenar(string campo)
    {
        if (Filtro.OrdenarPor == campo)
        {
            Filtro.OrdenDescendente = !Filtro.OrdenDescendente;
        }
        else
        {
            Filtro.OrdenarPor = campo;
            Filtro.OrdenDescendente = true;
        }

        Filtro.Pagina = 1;
        CargarDatos();
    }

    private void ManejarTecla(KeyboardEventArgs args)
    {
        if (args.Key == "Enter") Consultar();
    }

    private static string Moneda(decimal valor) =>
        valor.ToString("C2", CultureInfo.GetCultureInfo("es-MX"));

    private static string AbreviarUuid(string uuid) =>
        uuid.Length <= 13 ? uuid : $"{uuid[..8]}…{uuid[^4..]}";

    private static string ClaseTipo(string tipo) => tipo switch
    {
        "I" => "invoice",
        "N" => "payroll",
        "E" => "credit-note",
        "T" => "transfer",
        "P" => "payment",
        _ => "unknown"
    };

    private static string ClaseClasificacion(string clasificacion) => clasificacion switch
    {
        "EMITIDO" => "issued",
        "RECIBIDO" => "received",
        "PROPIO" => "own",
        _ => "unidentified"
    };

    private static string ObtenerUrlXml(long cfdiId)
    {
        return $"/api/cfdi/{cfdiId}/xml";
    }

    private static string ObtenerUrlPdf(long cfdiId)
    {
        return $"/api/cfdi/{cfdiId}/pdf";
    }
}
