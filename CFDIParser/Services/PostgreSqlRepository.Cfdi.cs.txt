    public long GuardarCfdi(
        long empresaId,
        CfdiDto cfdi,
        string clasificacion,
        string rutaXml)
    {
        ArgumentNullException.ThrowIfNull(cfdi);

        if (empresaId <= 0)
        {
            throw new InvalidOperationException(
                "La empresa seleccionada no es válida.");
        }

        if (string.IsNullOrWhiteSpace(cfdi.UUID))
        {
            throw new InvalidOperationException(
                "El CFDI no contiene un UUID válido.");
        }

        if (string.IsNullOrWhiteSpace(cfdi.RFCEmisor))
        {
            throw new InvalidOperationException(
                $"El CFDI {cfdi.UUID} no contiene RFC del emisor.");
        }

        if (string.IsNullOrWhiteSpace(cfdi.RFCReceptor))
        {
            throw new InvalidOperationException(
                $"El CFDI {cfdi.UUID} no contiene RFC del receptor.");
        }

        string clasificacionNormalizada =
            NormalizarClasificacion(clasificacion);

        using var conn =
            new NpgsqlConnection(_connectionString);

        conn.Open();

        using var tx =
            conn.BeginTransaction();

        try
        {
            long? cfdiExistenteId =
                BuscarCfdiId(
                    conn,
                    tx,
                    empresaId,
                    cfdi.UUID);

            long emisorId =
                ObtenerOCrearPersona(
                    conn,
                    tx,
                    cfdi.RFCEmisor,
                    cfdi.NombreEmisor,
                    "EMISOR");

            long receptorId =
                ObtenerOCrearPersona(
                    conn,
                    tx,
                    cfdi.RFCReceptor,
                    cfdi.NombreReceptor,
                    "RECEPTOR");

            long cfdiId;
            bool cfdiExistia =
                cfdiExistenteId.HasValue;

            if (cfdiExistia)
            {
                cfdiId =
                    cfdiExistenteId!.Value;

                ActualizarCfdi(
                    conn,
                    tx,
                    cfdiId,
                    empresaId,
                    emisorId,
                    receptorId,
                    cfdi,
                    clasificacionNormalizada,
                    rutaXml);

                EliminarImpuestosGlobales(
                    conn,
                    tx,
                    cfdiId);

                EliminarCfdiRelacionados(
                    conn,
                    tx,
                    cfdiId);
            }
            else
            {
                cfdiId =
                    InsertarCfdi(
                        conn,
                        tx,
                        empresaId,
                        emisorId,
                        receptorId,
                        cfdi,
                        clasificacionNormalizada,
                        rutaXml);
            }

            GuardarImpuestosGlobales(
                conn,
                tx,
                cfdiId,
                cfdi.Impuestos);

            GuardarCfdiRelacionados(
                conn,
                tx,
                cfdiId,
                cfdi.Relacionados);

            tx.Commit();

            return cfdiId;
        }
        catch
        {
            tx.Rollback();
            throw;
        }
    }

    public bool ExisteCfdi(
        long empresaId,
        string uuid)
    {
        if (empresaId <= 0 ||
            string.IsNullOrWhiteSpace(uuid))
        {
            return false;
        }

        using var conn =
            new NpgsqlConnection(_connectionString);

        conn.Open();

        const string sql = @"
            SELECT EXISTS
            (
                SELECT 1
                FROM cfdi
                WHERE empresa_id = @empresa_id
                  AND UPPER(TRIM(uuid)) =
                      UPPER(TRIM(@uuid))
            );";

        using var cmd =
            new NpgsqlCommand(
                sql,
                conn);

        cmd.Parameters.AddWithValue(
            "@empresa_id",
            empresaId);

        cmd.Parameters.AddWithValue(
            "@uuid",
            uuid.Trim());

        return Convert.ToBoolean(
            cmd.ExecuteScalar());
    }

    public long? ObtenerCfdiId(
        long empresaId,
        string uuid)
    {
        if (empresaId <= 0 ||
            string.IsNullOrWhiteSpace(uuid))
        {
            return null;
        }

        using var conn =
            new NpgsqlConnection(_connectionString);

        conn.Open();

        const string sql = @"
            SELECT cfdi_id
            FROM cfdi
            WHERE empresa_id = @empresa_id
              AND UPPER(TRIM(uuid)) =
                  UPPER(TRIM(@uuid))
            LIMIT 1;";

        using var cmd =
            new NpgsqlCommand(
                sql,
                conn);

        cmd.Parameters.AddWithValue(
            "@empresa_id",
            empresaId);

        cmd.Parameters.AddWithValue(
            "@uuid",
            uuid.Trim());

        object? resultado =
            cmd.ExecuteScalar();

        if (resultado is null ||
            resultado == DBNull.Value)
        {
            return null;
        }

        return Convert.ToInt64(resultado);
    }

            new NpgsqlConnection(_connectionString);

        conn.Open();

        const string sql = @"
            UPDATE cfdi
            SET monto_total_pagos =
                @monto_total_pagos
            WHERE empresa_id = @empresa_id
              AND UPPER(TRIM(uuid)) =
                  UPPER(TRIM(@uuid));";

    private static void GuardarImpuestosGlobales(
        NpgsqlConnection conn,
        NpgsqlTransaction tx,
        long cfdiId,
        IEnumerable<ImpuestoDto>? impuestos)
    {
        if (impuestos is null)
        {
            return;
        }

        const string sql = @"
            INSERT INTO cfdi_impuestos_globales
            (
                cfdi_id,
                tipo,
                impuesto,
                base,
                tipo_factor,
                tasa,
                importe
            )
            VALUES
            (
                @cfdi_id,
                @tipo,
                @impuesto,
                @base,
                @tipo_factor,
                @tasa,
                @importe
            )
            ON CONFLICT DO NOTHING;";

        foreach (ImpuestoDto impuesto in impuestos)
        {
            using var cmd =
                new NpgsqlCommand(
                    sql,
                    conn,
                    tx);

            cmd.Parameters.AddWithValue(
                "@cfdi_id",
                cfdiId);

            cmd.Parameters.AddWithValue(
                "@tipo",
                NormalizarTipoImpuesto(
                    impuesto.Tipo));

            AgregarParametro(
                cmd,
                "@impuesto",
                impuesto.Impuesto);

            cmd.Parameters.AddWithValue(
                "@base",
                impuesto.Base);

            AgregarParametro(
                cmd,
                "@tipo_factor",
                impuesto.TipoFactor);

            cmd.Parameters.AddWithValue(
                "@tasa",
                impuesto.Tasa);

            cmd.Parameters.AddWithValue(
                "@importe",
                impuesto.Importe);

            cmd.ExecuteNonQuery();
        }
    }

    private static void EliminarImpuestosGlobales(
        NpgsqlConnection conn,
        NpgsqlTransaction tx,
        long cfdiId)
    {
        const string sql = @"
            DELETE FROM cfdi_impuestos_globales
            WHERE cfdi_id = @cfdi_id;";

        using var cmd =
            new NpgsqlCommand(
                sql,
                conn,
                tx);

        cmd.Parameters.AddWithValue(
            "@cfdi_id",
            cfdiId);

        cmd.ExecuteNonQuery();
    }

    private static void GuardarCfdiRelacionados(
        NpgsqlConnection conn,
        NpgsqlTransaction tx,
        long cfdiId,
        IEnumerable<CfdiRelacionadoDto>? relacionados)
    {
        if (relacionados is null)
        {
            return;
        }

        const string sql = @"
            INSERT INTO cfdi_relacionados
            (
                cfdi_id,
                tipo_relacion,
                uuid_relacionado
            )
            VALUES
            (
                @cfdi_id,
                @tipo_relacion,
                @uuid_relacionado
            )
            ON CONFLICT DO NOTHING;";

        foreach (
            CfdiRelacionadoDto relacionado
            in relacionados)
        {
            if (string.IsNullOrWhiteSpace(
                    relacionado.UUID))
            {
                continue;
            }

            using var cmd =
                new NpgsqlCommand(
                    sql,
                    conn,
                    tx);

            cmd.Parameters.AddWithValue(
                "@cfdi_id",
                cfdiId);

            cmd.Parameters.AddWithValue(
                "@tipo_relacion",
                relacionado.TipoRelacion ?? "");

            cmd.Parameters.AddWithValue(
                "@uuid_relacionado",
                relacionado.UUID
                    .Trim()
                    .ToUpperInvariant());

            cmd.ExecuteNonQuery();
        }
    }

    private static void EliminarCfdiRelacionados(
        NpgsqlConnection conn,
        NpgsqlTransaction tx,
        long cfdiId)
    {
        const string sql = @"
            DELETE FROM cfdi_relacionados
            WHERE cfdi_id = @cfdi_id;";

        using var cmd =
            new NpgsqlCommand(
                sql,
                conn,
                tx);

        cmd.Parameters.AddWithValue(
            "@cfdi_id",
            cfdiId);

        cmd.ExecuteNonQuery();
    }

    private static string NormalizarClasificacion(
        string clasificacion)
    {
        string valor =
            (clasificacion ?? "")
                .Trim()
                .ToUpperInvariant();

        if (valor != "EMITIDO" &&
            valor != "RECIBIDO" &&
            valor != "PROPIO")
        {
            throw new InvalidOperationException(
                "La clasificación debe ser " +
                "EMITIDO, RECIBIDO o PROPIO.");
        }

        return valor;
    }

    private static string NormalizarTipoImpuesto(
        string tipo)
    {
        string valor =
            (tipo ?? "")
                .Trim()
                .ToUpperInvariant();

        return valor switch
        {
            "TRASLADO" =>
                "TRASLADO",

            "RETENCION" =>
                "RETENCION",

            "RETENCIÓN" =>
                "RETENCION",

            _ =>
                valor
        };
    }
